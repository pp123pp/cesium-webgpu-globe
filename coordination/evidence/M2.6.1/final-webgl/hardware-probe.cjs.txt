const {chromium} = require('@playwright/test');
const path = require('node:path');
(async()=>{
 const args = ['--enable-webgl','--use-gl=angle','--use-angle=d3d11','--ignore-gpu-blocklist','--disable-software-rasterizer','--enable-gpu-rasterization'];
 const browser = await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe',args,timeout:30000});
 try{
   const page = await browser.newPage();
   const result = await page.evaluate(async () => {
     const c=document.createElement('canvas');
     const gl=c.getContext('webgl2',{failIfMajorPerformanceCaveat:true}) || c.getContext('webgl',{failIfMajorPerformanceCaveat:true});
     if(!gl) throw new Error('Hardware WebGL creation failed');
     const ext=gl.getExtension('WEBGL_debug_renderer_info');
     return {version:gl.getParameter(gl.VERSION),vendor:gl.getParameter(ext ? ext.UNMASKED_VENDOR_WEBGL:gl.VENDOR),renderer:gl.getParameter(ext ? ext.UNMASKED_RENDERER_WEBGL:gl.RENDERER),extPresent:!!ext,userAgent:navigator.userAgent};
   });
   const raw=JSON.stringify({flags:args,...result});
   console.log('HARDWARE_WEBGL_PROBE='+raw);
   if(/swiftshader|software|llvmpipe|warp|basic render|microsoft basic/i.test(result.renderer))throw new Error('Software adapter not permitted: '+result.renderer);
   if(!/Intel|NVIDIA|AMD|Radeon|Arc\b/i.test(result.renderer))throw new Error('No positive physical adapter identity: '+result.renderer);
   console.log('PHYSICAL_GPU_VERIFIED');
 }finally{await browser.close();}
})().catch(e=>{console.error(e.stack);process.exitCode=1;});
